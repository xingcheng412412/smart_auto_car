cmake_minimum_required(VERSION 3.8)
project(fw_mini_pro_slam3d)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
add_library(cloud_conversion src/cloud_conversion.cpp)
target_include_directories(cloud_conversion PUBLIC include)
ament_target_dependencies(cloud_conversion sensor_msgs)
add_executable(fairy_adapter src/fairy_adapter_node.cpp)
target_link_libraries(fairy_adapter cloud_conversion)
ament_target_dependencies(fairy_adapter rclcpp sensor_msgs)
add_executable(fairy_map src/map_accumulator_node.cpp)
target_include_directories(fairy_map PRIVATE include)
ament_target_dependencies(fairy_map rclcpp sensor_msgs std_srvs)
install(TARGETS cloud_conversion fairy_adapter fairy_map
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ../config ../launch ../rviz ../scripts DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_voxel_map test/test_voxel_map.cpp)
  target_include_directories(test_voxel_map PRIVATE include)
  ament_add_gtest(test_cloud_conversion test/test_cloud_conversion.cpp)
  target_link_libraries(test_cloud_conversion cloud_conversion)
  ament_target_dependencies(test_cloud_conversion sensor_msgs)
endif()
ament_package()
